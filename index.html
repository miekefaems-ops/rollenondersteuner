<script>


document.getElementById('quizForm').addEventListener('submit', function(e) {


  e.preventDefault();




  const roleScores = {


    inclusiebezieler: 0,


    verbinder: 0,


    procesbewaker: 0,


    hulplijn: 0,


    professional: 0


  };




  const answers = document.querySelectorAll('input[type=radio]:checked');


  answers.forEach(answer => { roleScores[answer.value]++; });




  const highestRole = Object.keys(roleScores).reduce((a, b) => roleScores[a] > roleScores[b] ? a : b);




  const roleLabels = {


    inclusiebezieler: 'Inclusiebezieler',


    verbinder: 'Verbinder',


    procesbewaker: 'Procesbewaker',


    hulplijn: 'Hulplijn',


    professional: 'Professional met eigen bagage'


  };




  let scoresHtml = '<h3>Aantal keuzes per rol:</h3><ul>';


  for (const role in roleScores) {


    scoresHtml += `<li>${roleLabels[role]}: ${roleScores[role]}</li>`;


  }


  scoresHtml += '</ul>';




  document.getElementById('result').innerHTML = `


    <h2>Jouw dominante rol is: ${roleLabels[highestRole]}</h2>


    <p>Op basis van je antwoorden neem jij het vaakst de rol van <strong>${roleLabels[highestRole]}</strong> op tijdens je ondersteuning.</p>


    ${scoresHtml}


  `;


});


</script>
